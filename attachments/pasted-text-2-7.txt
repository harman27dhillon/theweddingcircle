function CeremonyCard({
  ceremony,
  featured,
  onOpen,
}: {
  ceremony: Ceremony;
  featured: boolean;
  onOpen?: (ceremony: Ceremony) => void;
}) {
  return (
    <article className={`pbc-card ${featured ? "pbc-card-featured" : ""}`}>
      <FloralCorner className="pbc-card-floral" />

      {featured && (
        <div className="pbc-up-next">
          <span>✦</span> UP NEXT
        </div>
      )}

      <div className="pbc-card-art">
        <CeremonyArtwork type={ceremony.artwork} />
      </div>

      <div className="pbc-card-content">
        <div className="pbc-eyebrow">
          {ceremony.id} <span>|</span> {ceremony.name.toUpperCase()}
        </div>

        <h3>{ceremony.name}</h3>

        <div className="pbc-event-meta">
          <span><CalendarDays /> {ceremony.date}</span>
          <i />
          <span><Clock3 /> {ceremony.time}</span>
        </div>

        <div className="pbc-location">
          <MapPin /> <span>{ceremony.location}</span>
        </div>

        <p className="pbc-description">{ceremony.description}</p>
      </div>

      <button
        className="pbc-arrow"
        aria-label={`View ${ceremony.name}`}
        onClick={() => onOpen?.(ceremony)}
      >
        <ChevronRight />
      </button>
    </article>
  );
}

type Props = {
  events?: Ceremony[];
  onOpenCeremony?: (ceremony: Ceremony) => void;
  onNavigate?: (tab: string) => void;
  activeTab?: string;
};

export default function PastelBlushCeremonies({
  events = ceremonies,
  onOpenCeremony,
  onNavigate,
  activeTab = "Home",
}: Props) {
  const next = events[0];
  const remaining = events.slice(1);

  return (
    <div className="pastel-blush-ceremonies">
      <header className="pbc-header">
        <div>
          <div className="pbc-couple-name">Surajpal &amp; Rekha</div>
          <div className="pbc-role"><span>●</span> Circle developer</div>
        </div>
        <div className="pbc-header-actions">
          <button aria-label="Notifications"><Bell /><span>40</span></button>
          <button aria-label="Host controls"><UserRound /><span>Host</span></button>
        </div>
      </header>

      <main className="pbc-main">
        <section className="pbc-intro">
          <FloralCorner className="pbc-intro-left" />
          <FloralCorner className="pbc-intro-right" />
          <h1>Our Celebrations</h1>
          <div className="pbc-divider"><span>✿</span></div>
          <p>A JOURNEY OF LOVE, FAMILY AND TOGETHERNESS</p>
        </section>

        {next && (
          <section className="pbc-list" aria-label="Wedding ceremonies">
            <CeremonyCard
              ceremony={next}
              featured
              onOpen={onOpenCeremony}
            />
            {remaining.map((event) => (
              <CeremonyCard
                key={event.id}
                ceremony={event}
                featured={false}
                onOpen={onOpenCeremony}
              />
            ))}
          </section>
        )}
      </main>

      <nav className="pbc-bottom-nav" aria-label="Wedding navigation">
        {[
          { name: "Home", Icon: Home },
          { name: "Chat", Icon: MessageCircle },
          { name: "Live", Icon: Radio },
          { name: "Album", Icon: Clapperboard },
          { name: "More", Icon: Menu },
        ].map(({ name, Icon }) => (
          <button
            key={name}
            className={activeTab === name ? "active" : ""}
            onClick={() => onNavigate?.(name)}
          >
            <span className="pbc-nav-icon"><Icon /></span>
            <span>{name}</span>
          </button>
        ))}
      </nav>
    </div>
  );
}
